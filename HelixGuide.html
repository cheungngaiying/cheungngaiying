        };

        const b = base[key] || { left: 40, bottom: 30 };

        const offsets = [

            { dx: 0, dy: 0 },

            { dx: -8, dy: 6 },

            { dx: 8, dy: -6 },

            { dx: -12, dy: -4 },

            positions.push({

                left: Math.max(1, Math.min(90, b.left + o.dx)) + '%',

                bottom: Math.max(1, Math.min(85, b.bottom + o.dy)) + '%'

            });

        }

        return positions;

    function startDrag(e) {

        e.preventDefault();

        const el = e.currentTarget;

        const id = el.dataset.id;

        const rect = el.getBoundingClientRect();

        const container = sceneContainer.getBoundingClientRect();

        const offsetX = e.clientX - rect.left;

        const offsetY = e.clientY - rect.top;

 

        dragData = { el, id, offsetX, offsetY, container };

        el.classList.add('dragging');

 

        document.addEventListener('mousemove', onDrag);

        document.addEventListener('mouseup', endDrag);

    }

 

    function startDragTouch(e) {

        e.preventDefault();

        const touch = e.touches[0];

        const el = e.currentTarget;

        const id = el.dataset.id;

        const rect = el.getBoundingClientRect();

        const container = sceneContainer.getBoundingClientRect();

        const offsetX = touch.clientX - rect.left;

        const offsetY = touch.clientY - rect.top;

 

        dragData = { el, id, offsetX, offsetY, container };

        el.classList.add('dragging');

 

        document.addEventListener('touchmove', onDragTouch, { passive: false });

        document.addEventListener('touchend', endDragTouch, { passive: false });

    }

 

    function onDrag(e) {

        if (!dragData) return;

        const { el, id, offsetX, offsetY, container } = dragData;

        const x = e.clientX - container.left - offsetX;

        const y = e.clientY - container.top - offsetY;

        const leftPct = Math.max(0, Math.min(100, (x / container.width) * 100));

        const bottomPct = Math.max(0, Math.min(100, (y / container.height) * 100));

        el.style.left = leftPct + '%';

        el.style.bottom = bottomPct + '%';

        itemPositions[id] = { left: leftPct + '%', bottom: bottomPct + '%' };

    }

 

    function onDragTouch(e) {

        e.preventDefault();

        if (!dragData) return;

        const touch = e.touches[0];

        const { el, id, offsetX, offsetY, container } = dragData;

        const x = touch.clientX - container.left - offsetX;

        const y = touch.clientY - container.top - offsetY;

        const leftPct = Math.max(0, Math.min(100, (x / container.width) * 100));

        const bottomPct = Math.max(0, Math.min(100, (y / container.height) * 100));

        el.style.left = leftPct + '%';

        el.style.bottom = bottomPct + '%';

        itemPositions[id] = { left: leftPct + '%', bottom: bottomPct + '%' };

    }

 

    function endDrag() {

        if (dragData) {

            dragData.el.classList.remove('dragging');

            dragData = null;

        }

        document.removeEventListener('mousemove', onDrag);

        document.removeEventListener('mouseup', endDrag);

    }

 

    function endDragTouch() {

        if (dragData) {

            dragData.el.classList.remove('dragging');

            dragData = null;

        }

        document.removeEventListener('touchmove', onDragTouch);

        document.removeEventListener('touchend', endDragTouch);

    }

    function keyboardHandler(e) {

        e.stopPropagation();

        const key = this.dataset.key;

        const errorEl = document.getElementById('codeError');

        if (key === 'backspace') {

            if (state.codeInput.length > 0) state.codeInput.pop();

            if (errorEl) errorEl.textContent = '';

        } else if (key === 'confirm') {

            const entered = state.codeInput.join('');

            if (entered.length !== state.codeLength) {

                if (errorEl) errorEl.textContent = '⚠️ Please enter exactly 6 letters';

                return;

            }

            if (state.savedCode && entered === state.savedCode) {

                showToastGlobal(`🔓 Correct code! Entering...`, 2000);

                state.codeInput = [];

                codeEntryOnly.classList.remove('active');

                mainLayout.classList.add('active');

                renderScene();

                updateLibraryStatus();

                updateChatFace();

                clearPopup();

                return;

            } else {

                if (errorEl) errorEl.textContent = '❌ Invalid code. Please try again.';

                state.codeInput = [];

                renderCodeEntryStandalone();

                return;

            }

        } else {

            if (state.codeInput.length < state.codeLength) {

                state.codeInput.push(key);

                if (errorEl) errorEl.textContent = '';

            } else {

                if (errorEl) errorEl.textContent = '⚠️ Maximum 6 letters';

            }

        }

        renderCodeEntryStandalone();

    }
